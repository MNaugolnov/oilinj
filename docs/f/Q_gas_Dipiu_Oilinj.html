Исходный код VBA · модуль Gas
Function Q_gas_Dipiu_Oilinj(rw As Single, re As Single, k As Single, kz As Single, pkr As Single, ppl As Single, pc As Single, zkr As Single, mol As Single, ro As Single, Tkr As Single, tpl As Single, s As Single, h As Single)
Dim rg As Single, zsr As Single, mu As Single, rspr As Single, Q_gas_Dipiu_Oilinj1 As Single
'rw- радиус скважины,  м 
'k- проницаемость, мД
'pkr - давление критическое, атм
' pc - забойное давление, атм
'ppl - пластовое давление, атм
' zkr - критический коэффициент сверхсжимаемости
'mol - молекулярная масса смеси  (для метана ­  16.24)
'ro - относительная плотность газа по воздуху
'tkr- критическая температура пласта, К
'tpl - температура пласта, К
's - скин-фактор
'h- эффективная толщина, м

zsr = Gas_koefz(Tkr, pkr, tpl, ((ppl + pc) / 2))
rg = rogas(pkr, Tkr, ro, ((ppl + pc) / 2), tpl)
mu = my(pkr, Tkr, mol, rg, tpl, ((ppl + pc) / 2), ro, zkr)
rspr = rw * Exp(-s)
k = k / 1000 * (10 ^ -12)
kz = kz / 1000 * (10 ^ -12)

ppl = ppl * 98066.5 '2026: ата -> Па
pc = pc * 98066.5
Q_gas_Dipiu_Oilinj1 = (3.14159265358979 * k * h * 293 * (ppl ^ 2 - pc ^ 2)) / mu / zsr / 101325# / tpl / Log(re / rspr)
Q_gas_Dipiu_Oilinj = Q_gas_Dipiu_Oilinj1 * 86.4

End Function
